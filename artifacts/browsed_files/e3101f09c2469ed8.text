  function eval5(cards){
    const ranks=cards.map(c=>c%13).sort((a,b)=>b-a); const suits=cards.map(c=>Math.floor(c/13));
    const counts=Array(13).fill(0); ranks.forEach(r=>counts[r]++);
    const flush=suits.every(s=>s===suits[0]); const uniq=[...new Set(ranks)].sort((a,b)=>b-a);
    let sh=-1; if(uniq.length===5){ if(uniq[0]-uniq[4]===4) sh=uniq[0]; else if(uniq[0]===12&&uniq[1]===3&&uniq[2]===2&&uniq[3]===1&&uniq[4]===0) sh=3; }
    const g=counts.map((n,r)=>({n,r})).filter(x=>x.n).sort((a,b)=>b.n-a.n||b.r-a.r);
    const pack=(cat,ks)=>{ const k=ks.slice(); while(k.length<5)k.push(0); return k.slice(0,5).reduce((s,x)=>s*13+x,cat); };
    if(flush&&sh>=0) return pack(sh===12?9:8,[sh]);
    if(g[0].n===4) return pack(7,[g[0].r,g[1].r]);
    if(g[0].n===3&&g[1]&&g[1].n===2) return pack(6,[g[0].r,g[1].r]);
    if(flush) return pack(5,ranks);
    if(sh>=0) return pack(4,[sh]);
    if(g[0].n===3) return pack(3,[g[0].r,g[1].r,g[2].r]);
    if(g[0].n===2&&g[1]&&g[1].n===2) return pack(2,[g[0].r,g[1].r,g[2].r]);
    if(g[0].n===2) return pack(1,[g[0].r,g[1].r,g[2].r,g[3].r]);
    return pack(0,ranks);
  }

  function apply(t, act){

    const alive=t.players.filter(x=>!x.folded);

    return t;
  }
  function street(t){
    const alive=t.players.filter(p=>!p.folded); if(alive.length<=1) return settle(t);
    t.players.forEach(p=>{ p.bet=0; p.acted=false; }); t.currentBet=0; t.minRaise=t.bb;
    if(t.street==="preflop"){ t.deck.pop(); t.board.push(t.deck.pop(),t.deck.pop(),t.deck.pop()); t.street="flop"; }

    return t;
  }
  function settle(t){
    t.street="showdown"; t.handOver=true; const alive=t.players.filter(p=>!p.folded);

    const sc=alive.map(p=>({p,...eval7(p.hole.concat(t.board))})).sort((a,b)=>b.score-a.score);
    const best=sc[0].score; const w=sc.filter(x=>x.score===best).map(x=>x.p); const award=t.pot-rake; const share=Math.floor(award/w.length);
    w.forEach(x=>x.stack+=share); t.winners={ids:w.map(x=>x.id),amount:award,label:HN[sc[0].cat]||"دست برتر"}; return t;
  }
  function botAct(t){
    const p=t.players[t.toAct]; const toCall=Math.max(0,t.currentBet-p.bet);
    const str = t.street==="preflop" ? Math.min(1, (Math.max(p.hole[0]%13,p.hole[1]%13)+ (p.hole[0]%13===p.hole[1]%13?6:0))/18) : eval7(p.hole.concat(t.board)).cat/9;
    if(toCall===0 && str<0.35) return {type:"check"};
    if(str>0.7){ if(p.stack>toCall) return {type:"raise", amount: Math.min(p.bet+p.stack, t.currentBet+Math.max(t.minRaise, Math.floor(t.pot*0.7)))}; return {type:"call"}; }
    if(toCall && str>0.28) return {type:"call"};
    if(toCall===0) return {type:"check"};
    return {type:"fold"};
  }

  function sit(id){
    const meta=TABLES.find(t=>t.id===id); if(S.user.chips<meta.min){ toast("ژتون کافی نیست"); return; }
    const stack=Math.min(S.user.chips, meta.max, Math.max(meta.min, meta.bb*80));
    S.buyin=stack; S.user.chips-=stack; S.user.cashable=Math.min(S.user.cashable,S.user.chips);
    const players=[{id:"hero",name:S.user.username,isHero:true,isBot:false,stack,hole:[],folded:false,allIn:false,bet:0,total:0,acted:false,style:"tight"}];
    BOTS.forEach((b,i)=>players.push({id:"bot"+i,name:b[0],isHero:false,isBot:true,stack:Math.floor(stack*(0.7+Math.random()*0.7)),hole:[],folded:false,allIn:false,bet:0,total:0,acted:false,style:b[1]}));
    S.table=startHand({id,name:meta.name,sb:meta.sb,bb:meta.bb,players,button:0,street:"idle",board:[],deck:[],pot:0,toAct:0,currentBet:0,minRaise:meta.bb,logs:[],winners:null,handOver:false,sessionRake:0});
    S.view="table"; render();
  }
  function leaveTable(){
    const hero=S.table.players.find(p=>p.isHero); const delta=hero.stack-S.buyin;
    S.user.chips+=hero.stack; if(delta>0) S.user.cashable+=Math.floor(delta*0.3); S.user.cashable=Math.min(S.user.cashable,S.user.chips);
    api("delta",{delta,note:"میز "+S.table.name,rake:S.table.sessionRake||0}).then(u=>S.user=u).catch(()=>{});
    S.table=null; S.view="lobby"; render();
  }
  function renderTable(){
    const t=S.table, hero=t.players.find(p=>p.isHero), others=t.players.filter(p=>!p.isHero);
    const actor=t.players[t.toAct], heroTurn=actor&&actor.isHero&&!t.handOver;
    const toCall=Math.max(0,t.currentBet-hero.bet);
    const app=document.getElementById("app");
    const seats=others.map((p,i)=>{
      const pos=SEATS[i]||SEATS[0];
      const on=t.players[t.toAct]===p&&!t.handOver;
      return `<div class="seat ${p.folded?"out":""} ${on?"on":""}" style="left:${pos.l};top:${pos.t}"><div>${p.name}</div><div class="muted">${fa(p.stack)}</div><div class="row" style="margin-top:4px">${t.handOver&&!p.folded?pc(p.hole[0],false,true)+pc(p.hole[1],false,true):pc(null,true,true)+pc(null,true,true)}</div>${p.bet?`<div class="chip">${fa(p.bet)}</div>`:""}</div>`;
    }).join("");
    app.innerHTML=`<div class="wrap" style="padding-bottom:8px">
      <div class="between"><button class="btn ghost" id="leave" style="width:auto">ترک میز</button><div class="muted">${t.name} · ${fa(t.sb)}/${fa(t.bb)}</div><div class="chip">${fa(hero.stack)}</div></div>
      <div class="felt-wrap"><div class="felt-outer"></div><div class="felt-inner"></div>${seats}
        <div style="position:absolute;left:50%;top:46%;transform:translate(-50%,-50%);text-align:center;z-index:2">
          <div class="muted">پات ${fa(t.pot)}</div>
          <div class="row" style="justify-content:center;margin-top:6px">${(t.board.length?t.board.map(c=>pc(c)).join(""):[0,1,2].map(()=>pc(null,true)).join(""))}</div>
          ${t.winners?`<div class="muted" style="margin-top:6px">${t.winners.label} · ${fa(t.winners.amount)}</div>`:""}
        </div>
      </div>
      <div class="between" style="margin-top:8px"><div class="row">${pc(hero.hole[0],!hero.hole.length)+pc(hero.hole[1],hero.hole.length<2)}</div><div class="muted">${t.logs.slice(-3).join("<br>")}</div></div>
      <div id="acts" style="margin-top:12px"></div>
    </div>`;
    document.getElementById("leave").onclick=leaveTable;
    const acts=document.getElementById("acts");
    if(t.handOver){
      acts.innerHTML=`<div class="grid2"><button class="btn sec" id="out">ترک میز</button><button class="btn" id="nx" ${hero.stack<t.bb?"disabled":""}>دست بعد</button></div>`;
      document.getElementById("out").onclick=leaveTable; document.getElementById("nx").onclick=()=>{ S.table=startHand(t); render(); };
    } else if(heroTurn){
      const minR=Math.min(hero.bet+hero.stack, t.currentBet+t.minRaise);

        <button class="btn" id="r">رِیز</button>
        <button class="btn sec" id="a">آل‌این</button></div>`;
      f.onclick=()=>{S.table=apply(t,{type:"fold"});render();};
      c.onclick=()=>{S.table=apply(t,{type:toCall?"call":"check"});render();};
      r.onclick=()=>{S.table=apply(t,{type:"raise",amount:minR});render();};
      a.onclick=()=>{S.table=apply(t,{type:"allin"});render();};
    } else acts.innerHTML=`<p class="muted" style="text-align:center">نوبت حریف…</p>`;
    if(!t.handOver && actor && actor.isBot){ setTimeout(()=>{ S.table=apply(S.table, botAct(S.table)); render(); }, 700+Math.random()*600); }
  }
