export function rakeAmount(pot: number, bb: number): number {
  const raw = Math.floor((pot * RAKE_BPS) / 10_000);
  const cap = bb * 10;
  return Math.min(raw, cap, Math.max(0, pot - 1));
}

export function cashoutToman(chips: number): { gross: number; fee: number; net: number } {
  const gross = Math.floor((chips / 1000) * SELL_TOMAN_PER_1000);
  const fee = Math.floor((gross * CASHOUT_FEE_BPS) / 10_000);
  return { gross, fee, net: gross - fee };
}

export function packageTotal(id: string) {
  const pack = PACKAGES.find((p) => p.id === id);
  if (!pack) return null;
  return { ...pack, total: pack.chips + pack.bonus };
}

export function winCashable(delta: number) {
  if (delta <= 0) return 0;
  return Math.floor((delta * WIN_CASHABLE_BPS) / 10_000);
}
